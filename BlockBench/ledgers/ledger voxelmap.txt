public VoxelShape makeShape(){
	VoxelShape shape = Shapes.empty();
	shape = Shapes.join(shape, Shapes.box(0, 0.8125, 0, 2, 1, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0, 0, 0.1875, 0.125, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0.125, 0.375, 0.1875, 0.8125, 0.625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.1875, 0, 0.375, 1.8125, 0.125, 0.625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 1, 0.125, 2, 1, 0.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.8125, 0, 0, 2, 0.125, 1), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(1.8125, 0.125, 0.375, 2, 0.8125, 0.625), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(2, 0.4375, 0.125, 2, 1, 0.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0, 0.4375, 0.125, 0, 1, 0.875), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.25, 1, 0.1875, 0.625, 1.0625, 0.75), BooleanOp.OR);
	shape = Shapes.join(shape, Shapes.box(0.1875, 1.0625, 0.25, 0.5625, 1.125, 0.8125), BooleanOp.OR);

	return shape;
}
